gnd += v_source['-']

  std::cerr << "Early initialization failed" << std::endl;

}

status = xyce.initializeLate();
if (status != Xyce::Circuit::Simulator::SUCCESS) {
  std::cerr << "Late initialization failed" << std::endl;
  return 1;
}

// Run the simulation
status = xyce.runSimulation();
if (status != Xyce::Circuit::Simulator::SUCCESS) {
  std::cerr << "Simulation failed" << std::endl;
  return 1;
}

The AscEditor class extends the functionality of SpiceEditor to work directly with LTSpice .asc schematic files. It allows for component value updates and simulation command injection before exporting to netlist or saving as a modified schematic.

```python
from PyLTSpice import AscEditor

# Load an LTSpice schematic file
schematic = AscEditor("./testfiles/sallenkey.asc")

# Apply modifications
schematic.set_component_value('R1', '10k')
schematic.set_parameters(Vos=0.001)

# Add simulation commands
schematic.add_instructions(".tran 0 10m 0 1u", ".meas TRAN vout_max MAX V(out)")

# Save as netlist or modified .asc
schematic.save_netlist("./testfiles/sallenkey_modified.net")
```

### Save Netlist to File using AscEditor

Source: https://github.com/nunobrum/pyltspice/blob/master/doc/classes/asc_editor.md

Saves the current circuit netlist to a specified file path. This method is part of the AscEditor class, which is designed for manipulating LTSpice .asc files. It takes the output file path as a parameter.

```python
from pathlib import Path
from PyLTSpice.editor import AscEditor

# Assuming 'asc_file_path' is the path to your .asc file
asc_file_path = Path("path/to/your/circuit.asc")

editor = AscEditor(asc_file_path)
editor.save_netlist("path/to/save/netlist.net")
```
